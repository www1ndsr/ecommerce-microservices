=== API through Gateway: register, login, catalog, cart, order and payment status ===
> $apiJob = Start-Job { kubectl port-forward -n shop svc/gateway 8000:8000 }
    try {
        Start-Sleep -Seconds 5
        $body = @{ email=$state.Email; password=$state.Pass } | ConvertTo-Json -Compress
        $registration = Invoke-WebRequest -Method Post -Uri 'http://localhost:8000/api/auth/register' -ContentType 'application/json' -Body $body -TimeoutSec 30
        $login = Invoke-WebRequest -Method Post -Uri 'http://localhost:8000/api/auth/login' -ContentType 'application/json' -Body $body -TimeoutSec 30
        $loginData = $login.Content | ConvertFrom-Json
        $state.Token = $loginData.access_token
        $headers = @{ Authorization=('Bearer '+$state.Token) }
        $identity = Invoke-WebRequest -Uri 'http://localhost:8000/api/auth/verify' -Headers $headers -TimeoutSec 30
        $state.UserId = ($identity.Content | ConvertFrom-Json).user_id
        $catalog = Invoke-WebRequest -Uri 'http://localhost:8000/api/products?limit=10' -TimeoutSec 30
        $cartBody = @{ product_id=1; quantity=1 } | ConvertTo-Json -Compress
        $cart = Invoke-WebRequest -Method Post -Uri 'http://localhost:8000/api/cart/items' -Headers $headers -ContentType 'application/json' -Body $cartBody -TimeoutSec 30
        $orderResponse = Invoke-WebRequest -Method Post -Uri 'http://localhost:8000/api/orders' -Headers $headers -TimeoutSec 30
        $orderData = $orderResponse.Content | ConvertFrom-Json
        $state.OrderId = $orderData.id; $state.OrderStatus = $orderData.status; $state.Total = $orderData.total
        $orderReadCode = 0
        for ($attempt=0; $attempt -lt 20 -and $state.OrderStatus -ne 'PAID'; $attempt++) {
            Start-Sleep -Seconds 2
            $orderRead = Invoke-WebRequest -Uri ('http://localhost:8000/api/orders/'+$state.OrderId) -Headers $headers -TimeoutSec 30
            $orderReadCode = $orderRead.StatusCode
            $orderData = $orderRead.Content | ConvertFrom-Json
            $state.OrderStatus = $orderData.status; $state.Total = $orderData.total
        }
        if ($state.OrderStatus -ne 'PAID') { throw ('Expected PAID, received '+$state.OrderStatus) }
        Write-Output ('Register HTTP '+$registration.StatusCode)
        Write-Output ('Login HTTP '+$login.StatusCode)
        Write-Output ('Catalog HTTP '+$catalog.StatusCode)
        Write-Output ('Cart HTTP '+$cart.StatusCode)
        Write-Output ('Create order HTTP '+$orderResponse.StatusCode)
        Write-Output ('Order GET HTTP '+$orderReadCode+'; ID='+$state.OrderId+'; STATUS='+$state.OrderStatus+'; TOTAL='+$state.Total)
    } finally {
        Stop-Job $apiJob -ErrorAction SilentlyContinue
        Remove-Job $apiJob -Force -ErrorAction SilentlyContinue
    }

=== Ten GET requests distributed across all three Order Service pods ===
> $podJson = Invoke-K @('get','pods','-n','shop','-l','app=order-service','-o','json')
    $podData = ($podJson -join "`n") | ConvertFrom-Json
    $podNames = @($podData.items | ForEach-Object { $_.metadata.name })
    if ($podNames.Count -ne 3) { throw ('Expected three order pods, found '+$podNames.Count) }
    $portBase = Get-Random -Minimum 16000 -Maximum 24000
    $jobs = @()
    try {
        for ($index=0; $index -lt $podNames.Count; $index++) {
            $portNumber = $portBase+$index
            $jobs += Start-Job -ArgumentList $podNames[$index],$portNumber -ScriptBlock { param($podName,$localPort) kubectl port-forward -n shop ('pod/'+$podName) ($localPort.ToString()+':8000') }
        }
        Start-Sleep -Seconds 5
        foreach ($job in $jobs) { if ($job.State -ne 'Running') { throw 'An Order Service port-forward did not start' } }
        for ($requestNumber=0; $requestNumber -lt 10; $requestNumber++) {
            $podIndex = $requestNumber % $podNames.Count; $portNumber = $portBase+$podIndex
            $request = Invoke-WebRequest -Uri ('http://localhost:'+$portNumber+'/api/orders') -Headers @{ 'X-User-ID'=[string]$state.UserId } -TimeoutSec 30
            Write-Output ('GET /api/orders #'+($requestNumber+1)+' -> HTTP '+$request.StatusCode+' via '+$podNames[$podIndex])
        }
        foreach ($podName in $podNames) { Write-Output ('Logs for '+$podName); Invoke-K @('logs',$podName,'-n','shop','--tail=5') }
    } finally {
        Stop-Job $jobs -ErrorAction SilentlyContinue
        Remove-Job $jobs -Force -ErrorAction SilentlyContinue
    }
